# project_gestion_asistencias_vacaciones_insip/docker-compose.yml

services:
  # Servicio de la Base de Datos SQL Server
  db:
    build:
      context: ./docker/sqlserver
      dockerfile: Dockerfile
    container_name: insip_sql_server
    restart: always
    environment:
      SA_PASSWORD: "${DB_PASSWORD}"
      ACCEPT_EULA: "Y"
    volumes:
      - sqlserver_data:/var/opt/mssql
      - ./db/init:/docker-entrypoint-initdb.d
      - ./db:/var/opt/mssql/backup
    ports:
      - "14330:1433"
    command: |
      /bin/bash -c "
        echo 'Iniciando SQL Server...'
        /opt/mssql/bin/sqlservr &
        echo 'Esperando a que SQL Server esté listo para aceptar conexiones y ejecutar scripts...'
        /docker-entrypoint-initdb.d/setup.sh
        echo 'SQL Server está listo y los scripts se han ejecutado.'
        wait
      "
    healthcheck:
      # AÑADIDO: Prefijo "CMD" como se requiere
      test: ["CMD", "/opt/mssql-tools/bin/sqlcmd", "-S", "localhost", "-U", "${DB_USERNAME}", "-P", "${DB_PASSWORD}", "-Q", "SELECT 1"]
      interval: 10s
      timeout: 10s
      retries: 20
      start_period: 90s

  # Servicio del Backend (API Laravel)
  backend_app:
    build:
      context: ./backend
      dockerfile: Dockerfile
    container_name: insip_backend_app
    restart: always
    volumes:
      - ./backend:/var/www/html
      - /var/www/html/vendor
    ports:
      - "9000:9000"
    environment:
      APP_NAME: ${APP_NAME}
      APP_ENV: ${APP_ENV}
      APP_KEY: ${APP_KEY}
      APP_DEBUG: ${APP_DEBUG}
      APP_URL: ${APP_URL}
      DB_CONNECTION: ${DB_CONNECTION}
      DB_HOST: db
      DB_PORT: ${DB_PORT}
      DB_DATABASE: ${DB_DATABASE}
      DB_USERNAME: ${DB_USERNAME}
      DB_PASSWORD: ${DB_PASSWORD}
      CACHE_DRIVER: ${CACHE_DRIVER}
      SESSION_DRIVER: ${SESSION_DRIVER}
      SESSION_LIFETIME: ${SESSION_LIFETIME}
      QUEUE_CONNECTION: ${QUEUE_CONNECTION}
      MAIL_MAILER: ${MAIL_MAILER}
      MAIL_HOST: ${MAIL_HOST}
      MAIL_PORT: ${MAIL_PORT}
      MAIL_USERNAME: ${MAIL_USERNAME}
      MAIL_PASSWORD: ${MAIL_PASSWORD}
      MAIL_ENCRYPTION: ${MAIL_ENCRYPTION}
      MAIL_FROM_ADDRESS: ${MAIL_FROM_ADDRESS}
      MAIL_FROM_NAME: ${MAIL_FROM_NAME}
      MAIL_ADMIN_RH: ${MAIL_ADMIN_RH}
      APP_FRONTEND_URL: ${APP_FRONTEND_URL}
      APP_FRONTEND_LOGIN_URL: ${APP_FRONTEND_LOGIN_URL}
      APP_FRONTEND_RESET_URL: ${APP_FRONTEND_RESET_URL}
      APP_FRONTEND_ACCOUNT_URL: ${APP_FRONTEND_ACCOUNT_URL}
      API_VERSION: ${API_VERSION}
      API_DESCRIPTION: ${API_DESCRIPTION}
      API_CONTACT_EMAIL: ${API_CONTACT_EMAIL}
      API_CONTACT_PHONE: ${API_CONTACT_PHONE}
      API_SUPPORT_URL: ${API_SUPPORT_URL}
      API_STATUS: ${API_STATUS}
      VITE_API_URL: ${VITE_API_URL}
    depends_on:
      db:
        condition: service_healthy
    command: >
      sh -c "
      echo 'Laravel esta listo para iniciar. Iniciando php-fpm....' &&
      php-fpm
      "

  # Servicio del servidor web (Nginx) para el Backend
  nginx_backend:
    image: nginx:alpine
    container_name: insip_nginx_backend
    restart: always
    ports:
      - "80:80"
    volumes:
      - ./backend:/var/www/html
      - ./docker/nginx/backend.conf:/etc/nginx/conf.d/default.conf
    depends_on:
      - backend_app
      
# NUEVO: Servicio del Queue Worker
  backend_worker:
    build:
      context: ./backend
      dockerfile: Dockerfile
    container_name: insip_backend_worker
    restart: always # Asegura que el worker siempre esté corriendo
    volumes:
      - ./backend:/var/www/html # Necesita acceder al código de la aplicación
      - /var/www/html/vendor
    environment: # Necesita las mismas variables de entorno que backend_app
      APP_ENV: ${APP_ENV}
      APP_KEY: ${APP_KEY}
      APP_DEBUG: ${APP_DEBUG}
      APP_URL: ${APP_URL}
      DB_CONNECTION: ${DB_CONNECTION}
      DB_HOST: db
      DB_PORT: ${DB_PORT}
      DB_DATABASE: ${DB_DATABASE}
      DB_USERNAME: ${DB_USERNAME}
      DB_PASSWORD: ${DB_PASSWORD}
      CACHE_DRIVER: ${CACHE_DRIVER}
      SESSION_DRIVER: ${SESSION_DRIVER}
      SESSION_LIFETIME: ${SESSION_LIFETIME}
      QUEUE_CONNECTION: ${QUEUE_CONNECTION} # Importante: debe ser 'database' o 'redis'
      MAIL_MAILER: ${MAIL_MAILER}
      MAIL_HOST: ${MAIL_HOST}
      MAIL_PORT: ${MAIL_PORT}
      MAIL_USERNAME: ${MAIL_USERNAME}
      MAIL_PASSWORD: ${MAIL_PASSWORD}
      MAIL_ENCRYPTION: ${MAIL_ENCRYPTION}
      MAIL_FROM_ADDRESS: ${MAIL_FROM_ADDRESS}
      MAIL_FROM_NAME: ${MAIL_FROM_NAME}
      MAIL_ADMIN_RH: ${MAIL_ADMIN_RH}
      APP_FRONTEND_URL: ${APP_FRONTEND_URL}
      APP_FRONTEND_LOGIN_URL: ${APP_FRONTEND_LOGIN_URL}
      APP_FRONTEND_RESET_URL: ${APP_FRONTEND_RESET_URL}
      APP_FRONTEND_ACCOUNT_URL: ${APP_FRONTEND_ACCOUNT_URL}
      API_VERSION: ${API_VERSION}
      API_DESCRIPTION: ${API_DESCRIPTION}
      API_CONTACT_EMAIL: ${API_CONTACT_EMAIL}
      API_CONTACT_PHONE: ${API_CONTACT_PHONE}
      API_SUPPORT_URL: ${API_SUPPORT_URL}
      API_STATUS: ${API_STATUS}
      VITE_API_URL: ${VITE_API_URL}
    depends_on:
      db:
        condition: service_healthy # El worker también necesita que la DB esté lista
    command: >
      sh -c "echo 'Esperando que la base de datos esté completamente lista antes de iniciar el Queue Worker...' && sleep 30 && echo 'Iniciando Laravel Queue Worker...' && php artisan queue:work --tries=3 --timeout=90 --sleep=3"

  # Servicio del Frontend (con su propio Nginx dentro del contenedor)
  frontend_app:
    build:
      context: ./frontend
      dockerfile: Dockerfile
      args:
        #pasamos las variables desde el .env pricipal
        VITE_API_URL_BUILD: ${VITE_API_URL}
        VITE_LAST_DEPLOYMENT_DATE_BUILD: ${VITE_LAST_DEPLOYMENT_DATE}
    container_name: insip_frontend_app
    restart: always
    ports:
      - "81:80"
    environment:
      - VITE_API_URL${VITE_API_URL}
      - VITE_LAST_DEPLOYMENT_DATE=${VITE_LAST_DEPLOYMENT_DATE}
    depends_on:
      - backend_app
